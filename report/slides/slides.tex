	\begin{frame}{Perturbation theory I}
		Perturbation theory is a mathematical approach that approximates solutions to problems that cannot be solved analytically, starting from the exact solution of a simpler related case.
		\vspace{3mm}
		
		\textbf{Base problem (unperturbed orbit): two bodies (Sun + Planet U)}
		\small
		\begin{align*}
			\text{(1) Gravitational Force:} \quad & \mathbf{F} = -\,G\,\frac{M_\odot m_2}{r^2}\,\hat{\mathbf{r}} \\[4pt]
			\text{(2) Newton’s Second Law:} \quad & \mathbf{F} = m_2\,\ddot{\mathbf{r}} \\[4pt]
			\text{Combine:} \quad & m_2 \ddot{\mathbf{r}} = -G\,\frac{M_\odot m_2}{r^2}\,\frac{\mathbf{r}}{r} \\[4pt]
			\Rightarrow \quad & \boxed{\ddot{\mathbf{r}} = -\frac{GM_\odot}{r^3}\,\mathbf{r}}
		\end{align*}	
		Solution: Keplerian ellipse with parameters $(a, e, i, \omega, \Omega, M)$.
	\end{frame}
	
	
	\begin{frame}{Perturbation theory II}
		
		\textbf{Perturbed problem (adding a third body): planet $P$ (mass $m'$ at $r'$)}
		\[
		\ddot{\mathbf r} = -\frac{G M}{r^3}\mathbf r + \epsilon\,\mathbf a_{\rm pert},\qquad
		\mathbf a_{\rm pert} = -G m'\left(\frac{\mathbf r-\mathbf r'}{|\mathbf r-\mathbf r'|^3} + \frac{\mathbf r'}{r'^3}\right)
		\]
		where $\epsilon \ll 1$ and $a_{\rm pert}$ represents the gravitational influence of another planet.
		
		\medskip
		Expanding orbital elements as:
		\[
		a(t) = a_0 + \epsilon a_1(t), \quad e(t) = e_0 + \epsilon e_1(t), \quad \ldots
		\]
		
		Substituting and keeping first-order terms in $\epsilon$:
		\[
		\frac{d\omega}{dt} = \frac{3}{4}n\frac{m'}{M_\odot}\left(\frac{a}{a'}\right)^3\!\!\frac{1}{(1 - e'^2)^{3/2}}
		\]
		\textit{This gives the secular precession of the perihelion, i.e., the slow change in the orientation of a celestial body's orbit, primarily caused by gravitational interactions with other bodies in the system.}
		
	\end{frame}
	
	\begin{frame}{Perturbation theory III}
		
		\textbf{Interpretation:}
		\Large
		
		Last equation can be just expressed as 
		\[
		\frac{d\omega}{dt} = C \, n \, \epsilon\left(\frac{a}{a'}\right)^3+\mathcal{O}(\epsilon^2)
		\]
		This means that a distant planet with small mass $m'$ produces a slow secular precession of Uranus’ perihelion (or a slow drift in mean longitude) proportional to $m'$ and $(a/a')^3$. Over decades this tiny rate integrates to an observable angular residual.
	\end{frame}

	\begin{frame}{Inverse perturbations}
		\Large
		Direct perturbations involve computing the effect of a known body, such as Jupiter’s influence on Uranus. However, Uranus’s orbit showed unexplained deviations.
		\vspace{3mm}
		
		Le Verrier and Adams deduced that these were caused by an unknown planet—Neptune. 
		\vspace{3mm}
		
		Inverse perturbation: the effect is observed but the cause (mass and position) must be inferred. Is much more complex than the direct case.
	\end{frame}

		\begin{frame}{Disturbing function and Lagrange planetary equations}
			Define disturbing function $\mathcal R(\mathbf r,\mathbf r')$ so that
			\[
			\mathbf a_{\rm pert} = \nabla_{\mathbf r}\mathcal R.
			\]
			Lagrange planetary equations give rates of orbital elements (mean motion $n= \sqrt{GM/a^3}$):
			\[
			\frac{da}{dt}=\frac{2}{n a}\frac{\partial\mathcal R}{\partial \lambda},\quad
			\frac{de}{dt}=\frac{\sqrt{1-e^2}}{n a^2 e}\frac{\partial\mathcal R}{\partial \omega} - \dots
			\]
			(where $\lambda$ is mean longitude, $\omega$ argument of perihelion).  (See standard texts / Laplace–Lagrange.)
		\end{frame}
		
		\begin{frame}{Inverse perturbation — strategy}
			\begin{enumerate}
				\item Compute residuals in Uranus' mean longitude $\Delta\lambda_{\rm obs}(t)$ vs model without unknown planet.
				\item Model secular response of $\lambda(t)$ to a distant perturber via analytical expansion of $\mathcal R$ (Legendre polynomial expansion) and orbit averaging.
				\item Fit parameters (mass $m'$, semimajor axis $a'$, mean longitude $\lambda'_0$) to minimize residuals (linearize if $m'$ small).
			\end{enumerate}
			This is what Le Verrier / Adams effectively did: match observed secular discrepancies to a plausible perturbing orbit.
		\end{frame}
		
		\begin{frame}{Sketch: expansion and secular term}
			For $r<r'$ expand:
			\[
			\frac{1}{|\mathbf r-\mathbf r'|} = \frac{1}{r'}\sum_{\ell=0}^{\infty}\left(\frac{r}{r'}\right)^\ell P_\ell(\cos\psi).
			\]
			Averaging over fast angles removes short-period terms; leading secular contribution scales like $(a/a')^\ell$. For an outer perturber the dominant secular contribution to perihelion precession scales like
			\[
			\dot{\varpi} \sim n\left(\frac{m'}{M}\right)\left(\frac{a}{a'}\right)^3 \times \mathcal{O}(1),
			\]
			so small $m'$ at moderate $a'$ can still produce measurable $\dot\varpi$ or $\Delta\lambda$ over decades.
		\end{frame}
